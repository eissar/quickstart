    function setClipboard(dat) {
  let str = Cc["@mozilla.org/supports-string;1"].createInstance(
    Ci.nsISupportsString
  );
  str.data = dat;
  let transferable = Cc["@mozilla.org/widget/transferable;1"].createInstance(Ci.nsITransferable);
  transferable.init(getLoadContext());
  transferable.addDataFlavor("text/plain");
  transferable.setTransferData("text/plain", str);
  Services.clipboard.setData(
    transferable,
    null,
    Ci.nsIClipboard.kGlobalClipboard
  );
}

    -- utils --------------------------------------------------------------
local function git_root(bufdir)
  -- Returns the absolute path of the git top‑level directory for *bufdir*,
  -- or nil if *bufdir* is not inside a repository.
  local out = vim.fn.systemlist('git -C ' .. vim.fn.fnameescape(bufdir) .. ' rev-parse --show-toplevel')
  if vim.v.shell_error ~= 0 then return nil end
  return out[1]
end

local function git_fetch(root)
  -- Run a cheap fetch (what `:Gfetch` does) – we ignore its output.
  vim.fn.systemlist('git -C ' .. vim.fn.fnameescape(root) .. ' fetch')
end

local function file_relative_to_root(filepath, root)
  -- Path of *filepath* as seen from *root* (git needs a relative path).
  return vim.fn.fnamemodify(filepath, ':.' .. root .. ':')
end

-- --------------------------------------------------------------------
--- Check whether the *current buffer* has any changes on the remote
--- counterpart (the upstream branch).
--- Returns `true` if there are remote changes, `false` otherwise.
function _G.has_remote_changes()
  local bufnr   = vim.api.nvim_get_current_buf()
  local path    = vim.api.nvim_buf_get_name(bufnr)

  if path == '' then                     -- no file name (e.g. a new buffer)
    return false
  end

  local bufdir  = vim.fn.fnamemodify(path, ':h')
  local root    = git_root(bufdir)
  if not root then                       -- not a git repo
    return false
  end

  -- Make sure we are up‑to‑date with the remote.
  git_fetch(root)

  -- Compare the file against the upstream of the current branch.
  -- `git diff --quiet @{u} -- <file>` exits with 0 if identical,
  -- non‑zero if there *are* differences.
  local rel = file_relative_to_root(path, root)
  local cmd = {
    'git', '-C', root, 'diff', '--quiet', '@{u}', '--', rel,
  }
  vim.fn.systemlist(cmd)                -- runs the command, discarding output
  return vim.v.shell_error ~= 0        -- non‑zero ⇒ remote changes exist
end

-- Optional convenience command ------------------------------------------------
vim.api.nvim_create_user_command(
  'CheckRemoteChanges',
  function()
    if _G.has_remote_changes() then
      vim.notify('⛔ Remote changes detected for the current file', vim.log.levels.WARN)
    else
      vim.notify('✅ No remote changes', vim.log.levels.INFO)
    end
  end,
  { desc = 'Check if the file in the current buffer differs from its upstream version' }
)
